\begin{exolab}{5.1}
  Let $X$ and $Y$ be topological spaces. Suppose $G$ is a group acting (continuously) on a topological space $X$, and let $\pi: X \to G \backslash X$ be the quotient map taking $x \mapsto Gx$.
  \begin{enumerate}[(a)]
    \item 
      Show that a map $f: G \backslash X \to Y$ is continuous if and only if $f \circ \pi: X \to Y$ is continuous and $G$-invariant, i.e., $(f \circ \pi)(gx) = (f \circ \pi)(x)$ for all $g \in G$ and $x \in X$.
    \item
      Show that there is a bijective correspondence
      \begin{equation*}
        \left\{
          \parbox{35mm}{
            \center
            continuous maps $f: G \backslash X \to Y$
          }
        \right\}
        \quad
        \overset{\sim}{\longleftrightarrow}
        \quad
        \left\{
          \parbox{45mm}{
            \center
            $G$-invariant continuous maps $h: X \to Y$
          }
        \right\} \, .
      \end{equation*}
  \end{enumerate}
\end{exolab}

\begin{exolab}{5.2}
  Let $\zeta = e^{2\pi i/r}$ be a primitive $r^\text{th}$ root of unity. Define $\sigma, \tau: \wh{\C} \to \wh{\C}$ by $z \mapsto \zeta z$ and $z \mapsto 1/z$, respectively.
  \begin{enumerate}[(a)]
    \item 
      Show that the subgroup $G \colonequals \langle \sigma, \tau \rangle$ of $\Aut(\wh{\C})$ is isomorphic to $D_{2r}$, the dihedral group of order $2r$.
    \item
      Let $\pi: \wh{\C} \to G \backslash \wh{\C}$ be the quotient map. Show that $\pi$ has $3$ ramification values whose corresponding ramification points have ramification indices $2,2,r$, respectively.
  \end{enumerate}
\end{exolab}
